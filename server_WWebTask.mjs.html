/**
 * 任務調度伺服器
 *
 * 身分驗證責任：框架（w-converhp、w-serv-hapi、w-serv-webdata、w-serv-orm）不驗證token，一律由本套件負責。連線閘（verifyConn）與執行身分（getUserIdByToken）皆須完整驗證（token有效、本系統使用者存在且啟用、部署方授權）；hapi apis路由各自驗證；函數層級另檢查角色或成員；公開入口只限列名者。
 * 本套件之落實：連線閘、執行身分（資料通道本體__sysToken__，from固定為'main'）、登入檢查共用同一個完整驗證（token認證＋verifyClientUser）；本套件無自有使用者表，「使用者存在且啟用」由getUserByToken查無即回null保證。
 * 函數層級（資料通道與HTTP REST共用同一張入口函數表，server/access.mjs）：管理類（頻道與成員之建立、修改、刪除）限系統管理者；訊息、任務、已讀游標、成員名冊限系統管理者或該頻道成員；檔案限系統管理者或任一頻道成員；頻道清單、統計、資料通道之channels／channelMembers的select只回呼叫者有權看之頻道之資料。
 * 系統管理者＝getUserByToken所回使用者之isAdmin嚴格等於'y'；應用系統token之使用者用途反查回所綁使用者（等級同其使用者），系統服務用途反查回id以'app:'開頭且isAdmin為'y'（等級為系統管理者）；本套件不分辨用途，不讀isApp。頻道成員＝channelMembers中(channelId, memberId為呼叫者id, isActive為'y')之列（agent若以一般使用者認證，其user.id須為成員）。
 * 公開入口（無須token）只有前端靜態檔；/api/*與資料通道一律須token，系統函數只公開'[sys:getFuncList]'（且本套件之getUserIdByToken對空token fail-closed）。
 *
 * @class
 * @param {Function} WOrm 輸入資料庫ORM函數
 * @param {String} url 輸入資料庫連線字串，例如w-orm-lmdb為'./db'
 * @param {String} db 輸入資料庫名稱字串
 * @param {Function} getUserByToken 輸入處理函數，函數會傳入使用者token，通過此函數處理後並回傳使用者資訊物件，並至少須提供'id'、'email'、'name'、'isAdmin'欄位，且'isAdmin'限輸入'y'或'n'；本套件以'isAdmin'為'y'者為系統管理者（角色之唯一來源，函數層級之管理類限此類使用者），其餘為一般使用者（只能存取自己為成員之頻道）
 * @param {Function} verifyClientUser 輸入驗證瀏覽使用者身份之處理函數，函數會傳入使用者資訊物件之淺拷貝與來源字串(from)，通過此函數識別後回傳布林值，允許使用者回傳true，反之回傳false。作用於登入檢查(/api/getUserByToken，from為'getUserByToken')與前端資料通道之每次請求(連線閘，from為w-converhp之apiType如'main'；執行身分每個函數呼叫前再驗一次，from為'main')，三者同一判定；拋錯或reject視同不允許。本套件於此函數通過後另自行要求角色與成員（見上），部署方只能在其上再收緊
 * @param {Function} verifyAppUser 輸入驗證應用程序使用者身份之處理函數，函數會傳入使用者資訊物件與來源字串(from，為API名稱)，通過此函數識別後回傳布林值，允許使用者回傳true，反之回傳false。作用於對外HTTP REST API(agent面)；拋錯或reject視同不允許。本套件於此函數通過後另自行要求角色與成員（管理類不在HTTP REST；頻道內容限系統管理者或頻道成員），部署方只能在其上再收緊
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Integer} [opt.serverPort=11008] 輸入伺服器通訊port，預設11008
 * @param {String} [opt.subfolder=''] 輸入網站子目錄字串，例如'mtask'，預設''
 * @param {String} [opt.urlRedirect=''] 輸入未登入時導向之網址字串(例如SSO登入頁)，預設''
 * @param {String} [opt.showLanguage='y'] 輸入是否顯示語系選單字串，可選'y'、'n'，預設'y'
 * @param {String} [opt.language='eng'] 輸入初始語系字串，可選'eng'、'cht'，預設'eng'
 * @param {Object} [opt.webName={}] 輸入網站名稱物件，key為語系而值為字串，預設{}
 * @param {Object} [opt.webDescription={}] 輸入網站說明物件，key為語系而值為字串，預設{}
 * @param {String} [opt.webLogo=''] 輸入網站logo字串，預設''
 * @param {Object} [opt.kpLangExt={}] 輸入擴充語系物件，預設{}
 * @param {String} [opt.logFd='./logs'] 輸入後端log資料夾字串，相對路徑以建構時之工作目錄解析，資料夾不可建立時(例如上層為檔案)於建構時拋錯(w-syslog 1.0.33起；於資料庫建立之前)，預設'./logs'
 * @param {String} [opt.logInterval='hr'] 輸入後端log分檔模式字串，可選'day'、'hr'，預設'hr'
 * @param {Integer} [opt.archiveAfterDays=30] 輸入終態任務與舊訊息逾此天數搬至冷表之天數，小於等於0為停用，預設30
 * 已移除之設定：useCheckUser、getUserById、useExcludeWhenNotAdmin(w-serv-orm已移除)，給予undefined、null、false以外之值即於建構時拋錯(於任何副作用之前)
 * @returns {Object} 回傳w-serv-hapi之伺服器事件物件，可監聽error事件；w-serv-hapi 1.1.6起另帶ready與stop，惟stop只停止伺服器與其下游，不釋放本套件之計時器(封存掃描、雙擊防護占位)與資料庫連線，本套件以結束行程收尾
 */
function WWebTask(WOrm, url, db, getUserByToken, verifyClientUser, verifyAppUser, opt = {}) {
    let instWServHapiServer = null


    //check WOrm
    if (!isfun(WOrm)) {
        console.log('invalid WOrm', WOrm)
        throw new Error('invalid WOrm')
    }


    //check url
    if (!isestr(url)) {
        console.log('invalid url', url)
        throw new Error('invalid url')
    }


    //check db
    if (!isestr(db)) {
        console.log('invalid db', db)
        throw new Error('invalid db')
    }


    //check getUserByToken
    if (!isfun(getUserByToken)) {
        console.log('invalid getUserByToken', getUserByToken)
        throw new Error('invalid getUserByToken')
    }


    //check verifyClientUser
    if (!isfun(verifyClientUser)) {
        console.log('invalid verifyClientUser', verifyClientUser)
        throw new Error('invalid verifyClientUser')
    }


    //check verifyAppUser
    if (!isfun(verifyAppUser)) {
        console.log('invalid verifyAppUser', verifyAppUser)
        throw new Error('invalid verifyAppUser')
    }


    //checkOptRemoved: w-serv-orm 已移除之 useCheckUser / getUserById / useExcludeWhenNotAdmin(spec D17), 本套件不讀不轉交;
    //給予 undefined / null / false 以外之值即同步拋 w-serv-orm 之原文訊息. 置於任何副作用(srLog 之 log 目錄、資料庫)之前; 規則與訊息由 w-serv-orm 單一擁有
    checkOptRemoved(opt)


    //serverPort
    let serverPort = get(opt, 'serverPort')
    if (!ispint(serverPort)) {
        serverPort = 11008
    }
    serverPort = cint(serverPort)


    //webName
    let webName = get(opt, 'webName', {})


    //webDescription
    let webDescription = get(opt, 'webDescription', {})


    //webLogo
    let webLogo = get(opt, 'webLogo', '')


    //subfolder
    let subfolder = get(opt, 'subfolder', '')
    if (isestr(subfolder)) {
        if (strright(subfolder, 1) === '/') { //右邊不需要給「/」
            subfolder = strdelright(subfolder, 1)
        }
        if (strleft(subfolder, 1) !== '/') { //左邊需要給「/」
            subfolder = `/${subfolder}`
        }
    }


    //urlRedirect
    let urlRedirect = get(opt, 'urlRedirect', '')


    //params
    let showLanguage = get(opt, 'showLanguage', 'y')
    let language = get(opt, 'language', 'eng')
    let logFd = get(opt, 'logFd', './logs') //srLog 寫入之 log 資料夾


    //kpLangExt
    let kpLangExt = get(opt, 'kpLangExt', {})


    //kpLang
    let kpLang = procLang({ kpLangExt, webName, webDescription })


    //srLog（後端 log；錯誤一律 log err-key 供識別）
    let srLog = srLogInit(opt)

    //注入函數一律經 wrapInjected 包裝後使用（W 契約，spec/設計要點與取捨.md D15）：
    //部署方注入之 getUserByToken / verifyClientUser / verifyAppUser 同步拋錯或 reject 時，視同否定結果（查無 → null、無權限 → false），
    //走各呼叫點既有分支回自家 key；失敗原文（舊版 sso helper 為已代入權杖之完整網址）不上拋、不入 log，只記 describeUpstream 之 key 或型別。
    //以包裝版取代參數本身 → 其後所有呼叫點（含日後新增者）結構上只可能用到包裝版；包裝處記 warn（因），呼叫點記 error（果）。
    let onInjectFail = (name) => {
        return (d) => {
            srLog.warn({ event: `inject-${name}-fail`, ...d })
        }
    }
    getUserByToken = wrapInjected(getUserByToken, { failValue: null, onFail: onInjectFail('getUserByToken') })
    verifyClientUser = wrapInjected(verifyClientUser, { failValue: false, onFail: onInjectFail('verifyClientUser') })
    verifyAppUser = wrapInjected(verifyAppUser, { failValue: false, onFail: onInjectFail('verifyAppUser') })


    //WServOrm
    let wp = {}
    try {
        wp = WServOrm(ds, WOrm, url, db)
    }
    catch (err) {
        //ORM 為必要依賴：初始化失敗即 fail-fast，不以 wp={} 帶 undefined woItems/procOrm 殘缺啟動
        console.log('WServOrm init failed', err)
        throw err
    }
    let { woItems, procOrm, withTabLocks, guardWrite } = wp //withTabLocks / guardWrite: 每表一把表鎖與「取鎖 → 新鮮讀取 → 基準比對 → 寫入」原子防護 (多人同時編輯, 原則 10; 同表之所有寫入者共用此實例之鎖)


    //印權杖（console）一律經 maskLog 之 maskTok / maskQuery（M 契約，含陣列與非字串），不另寫第二份遮罩

    //runApi: 統一 HTTP handler 收尾 — pm2resolve + isErr 還原 + 失敗記 srLog.error (與 kpFunExt 之 wrapErrLog 對稱)
    let runApi = async (event, core) => {
        let r = await pm2resolve(core)()
        if (isErr(r.msg)) {
            r.msg = r.msg.message
        }
        if (r.state !== 'success') {
            srLog.error({ event: 'api/' + event, err: r.msg })
        }
        return r
    }

    //kpFunExt 邊界統一處理：成功記 info；catch 後端錯誤 → srLog 該 err-key → 原樣 re-reject。
    //event 統一加 'kpfun-' 前綴 → 專用事件、與 api/verifyConn 區分。
    let wrapErrLog = (event, fn) => {
        let ev = 'kpfun-' + event
        return async (...args) => {
            try {
                let result = await fn(...args)
                srLog.info({ event: ev }) //成功也記 info
                return result
            }
            catch (err) {
                srLog.error({ event: ev, err })
                return Promise.reject(err)
            }
        }
    }


    //kp: 資料通道函數(kpFunExt)的呼叫者轉接: 由 userId 查回 getUserIdByToken 驗證當下登錄之角色(access.callerOf; 無登錄或逾期即拒絕), 再交給 gf 之函數(規則檢查在 gf 內), 並經 wrapErrLog
    let kp = (name, fn) => {
        return wrapErrLog(name, async (userId, ...args) => {
            let caller = await access.callerOf(userId)
            return await fn(caller, ...args)
        })
    }


    //kmx: per-key in-memory mutex, 同 key 序列化、不同 key 並行. 任務生命週期 critical 路徑用.
    let kmx = pmKeyMutex()


    //lockSave: 有副作用之寫入路徑之雙擊防護(後端, 見 lockSave.mjs, spec D16): 同一操作者之同一操作(或同一列)處理中再送出即 reject, 不排隊.
    //cacheSt 內含 TTL 偵測 timer(setInterval) → 建於此並注入 procCore, 不於 procCore 內建立: procCore 保持無 timer, unit 測試可獨立建構退出(同 D12 封存 timer 之安排)
    let cst = cacheSt()
    let lockSave = createLockSave(cst)


    //procCore
    let pc = procCore({ woItems, procOrm, ds, srLog, kmx, lockSave, withTabLocks, guardWrite })


    //access: 呼叫者角色與成員檢查之單一擁有者 (server/access.mjs; 規格見 spec/設計要點與取捨.md D20). procCore 不做授權, 授權只在下方 gf (入口函數表) 套用一次.
    let access = createAccess({ woItems, srLog })


    //封存掃描 (spec D12 方向②: 冷熱分離): 啟動後 30s 首掃 + 每小時掃一次, 將過齡終態任務/舊訊息搬至冷表.
    //timer 註冊於此 (不放 procCore) → procCore 保持無 timer, unit 測試可獨立建構與退出.
    //archiveAfterDays<=0 停用 (archiveSweep 內亦有守衛).
    let archiveAfterDays = cint(get(opt, 'archiveAfterDays', 30))
    if (archiveAfterDays > 0) {
        let lockingForArchive = false
        let runArchiveSweep = async () => {
            if (lockingForArchive) {
                return
            }
            lockingForArchive = true
            await pc.archiveSweep(archiveAfterDays)
                .then((r) => {
                    if (r.tasksMoved > 0 || r.messagesMoved > 0) {
                        srLog.info({ event: 'archiveSweep', tasksMoved: r.tasksMoved, messagesMoved: r.messagesMoved })
                    }
                })
                .catch((err) => {
                    srLog.error({ event: 'archiveSweep-error', msg: (typeof err === 'string') ? err : ((err && err.message) || 'unknown') })
                })
                .finally(() => {
                    lockingForArchive = false
                })
        }
        setTimeout(runArchiveSweep, 30 * 1000)
        setInterval(runArchiveSweep, 60 * 60 * 1000)
    }


    //getWebInfor
    let getWebInfor = (userId) => {
        return {

            // webName, //已併入kpLang
            // webDescription, //已併入kpLang
            webLogo,

            showLanguage,
            language,
            kpLang,

        }
    }


    //gf: 全部對外入口函數 (資料通道 kpFunExt、對外 HTTP REST 共用) 與其呼叫者規則之單一表, 每個函數收 (caller, ...args):
    //管理類限系統管理者; 頻道內容 (訊息、任務、游標、成員名冊) 限系統管理者或該頻道成員; 檔案限系統管理者或任一頻道成員; 清單與統計只回有權看之頻道之資料 (見 access.mjs)
    let gf = createGuardedFuncs({ access, pc, getWebInfor })


    //pickUser: 使用者物件之對外 / 對 console 投影 (白名單 id / name / email / isAdmin, 前端 userSelf 只讀此四欄): 部署方 getUserByToken 回傳之物件可能夾帶其他欄位 (如密鑰、內部識別碼),
    //登入檢查之回應與認證失敗之 console 皆只出此投影 (同 w-web-api A-2; 外審 sol#3 / opus#9)
    let pickUser = (user) => {
        return {
            id: get(user, 'id', ''),
            name: get(user, 'name', ''),
            email: get(user, 'email', ''),
            isAdmin: get(user, 'isAdmin', ''),
        }
    }


    //checkUser
    let checkUser = async(user) => {
        let id = get(user, 'id', '')
        if (!isestr(id)) {
            console.log('user', pickUser(user))
            console.log('can not get the userId')
            return Promise.reject('errUserIdMissing')
        }
        let email = get(user, 'email', '')
        if (!isestr(email)) {
            console.log('user', pickUser(user))
            console.log('can not get the email of user')
            return Promise.reject('errUserEmailMissing')
        }
        let name = get(user, 'name', '')
        if (!isestr(name)) {
            console.log('user', pickUser(user))
            console.log('can not get userName')
            return Promise.reject('errUserNameMissing')
        }
        let isAdmin = get(user, 'isAdmin', '')
        if (isAdmin !== 'y' && isAdmin !== 'n') {
            console.log('user', pickUser(user))
            console.log('user.isAdmin is not y or n', user.isAdmin)
            console.log('can not get the role of user')
            return Promise.reject('errUserRoleMissing')
        }
        return true
    }


    //getTokenUser：token → 使用者之唯一出口（client 路徑、app 路徑、連線閘 p.checkToken 共用）。
    //查無（含上游失敗經 wrapInjected 視同查無）一律 errUserNotFound（S7；與 perm getTokenUser 一致），欄位不齊再由 checkUser 回對應 key。
    let getTokenUser = async (token) => {

        //user
        let user = null
        if (isestr(token)) {
            user = getUserByToken(token)
            if (ispm(user)) {
                user = await user
            }
        }

        //check
        if (!iseobj(user)) {
            console.log(`token`, maskTok(token))
            console.log(`can not find the user from token`)
            return Promise.reject('errUserNotFound')
        }

        //checkUser: client 與 app 路徑對稱 (id/email/name/isAdmin 缺一即拒).
        //agent 建議以 SSO app token (isApp='y') 認證 (使用者用途反查回所綁使用者之實際資料; 系統服務用途反查回 id='app:<金鑰主鍵>' 與非空占位之 name / email, isAdmin='y'), 四欄皆為非空值, 不受本檢查影響.
        await checkUser(user)

        return user
    }


    //authToken：瀏覽使用者 token 完整驗證之單一擁有者（連線閘 p.checkToken、執行身分 getUserIdByToken、登入檢查 getAndVerifyClientUser 三者共用；C3，
    //dspec 使用者身分驗證原則.md 二.1、二.2：執行身分即驗證身分之步驟，須與連線閘同等完整，不可只查 token 對應之 id）。
    //認證：getTokenUser（解析呼叫者、欄位齊備；本套件無自有使用者表，「使用者存在且啟用」由部署方 getUserByToken 查無即回 null 保證，查無一律 errUserNotFound）；
    //授權：已包裝之 verifyClientUser（拋錯或 reject 視同 false 並記 inject warn），from 為入口名（連線閘為 w-converhp 之 apiType，執行身分固定為 'main'，登入檢查為 'getUserByToken'）。
    //回 { user, allowed }：認證失敗 reject 自家 key；授權不通過回 allowed:false，由呼叫端依其入口記 log 與回應。
    //交給鉤子者為淺拷貝：鉤子改寫傳入之物件（如把 isAdmin 改為 'y'）不影響其後之角色判斷（角色取自 access.remember／callerOfUser 所見之原物件）。
    let authToken = async (token, from = '') => {

        //getTokenUser

        },
        getUserIdByToken: async (token) => { //可使用async或sync函數
            //執行身分（C3，spec D20）：w-serv-webdata 於執行每個資料通道函數（kpFunExt、表 select、系統函數）之前以請求本體之 __sysToken__ 呼叫本函數取得 userId，
            //此 token 與連線閘所驗之 Authorization 標頭 token 來自不同欄位、可為不同值（框架不比對兩者）。故「反查使用者」即驗證身分之步驟，須與連線閘同等完整：
            //token 認證（解析、欄位齊備）＋部署方授權 verifyClientUser(user, 'main')，共用 authToken（from 固定為 'main'：資料通道函數只由 /api/main 之 execute 觸發，與連線閘 'main' 為同一入口）。
            //fail-closed（S9）：token 非字串、查無、欄位不齊、授權不通過（含上游失敗經 wrapInjected 視同查無／否定）一律 reject 'errUserIdMissing'，函數不執行；
            //原因只入 srLog（err 為自家 key，不含 token 與使用者資料）。通過者登錄其角色供其後同一呼叫之函數層級檢查（access.remember／callerOf；見 access.mjs）。
            let seq = access.nextSeq() //驗證開始時取序號: 同一 userId 並行驗證之完成順序可能顛倒, 登錄只接受不舊之序號 (access.remember)
            let r = null
            try {
                r = await authToken(token, 'main')
            }
            catch (err) {
                console.log('token', maskTok(token))
                console.log('can not get the userId from token')
                srLog.error({ event: 'getUserIdByToken', err: 'errUserIdMissing' })
                return Promise.reject('errUserIdMissing')
            }
            if (!r.allowed) {
                console.log('token', maskTok(token))
                console.log('user does not have permission for execution')
                srLog.error({ event: 'getUserIdByToken', err: 'errUserNoPermission' })
                return Promise.reject('errUserIdMissing')
            }
            access.remember(r.user, seq)
            return get(r.user, 'id', '') //getTokenUser／checkUser 已驗 id 為非空字串
        },
        //系統函數收緊（第 128 項；dspec 使用者身分驗證原則.md 二.7、二.8）：只公開 [sys:getFuncList]（函數名稱清單，無敏感內容）；
        //[sys:polling]（取走廣播與同步表時間戳）與 [sys:getTableTags]（同步表名與時間戳）須有效身分——輪詢之 __sysToken__ 由前端元件（w-serv-hapi 1.1.8／w-serv-webdata 1.1.16／w-serv-broadcast 1.1.12 起）每輪帶上，經 getUserIdByToken 反查。
        //本套件前端與後端同一套件發布，前端建置檔已用新元件，故可顯式開啟（原則二.8）；已發布之舊前端建置檔輪詢不帶 token，會被拒（需重新載入）。
        sysFuncsPublic: ['[sys:getFuncList]'],
        useDbOrm: true,
        kpOrm: woItems,
        //operOrm：表 select 供前端同步 channels / channelMembers（authorizeExec 只能整次放行或拒絕）；一般使用者只得所屬頻道之列（C2b，access.scopeSelect），系統管理者全部。
        //先確認呼叫者登錄（getUserIdByToken 驗證當下寫入，逾期或無則拒絕）再讀取，不讀了才拒絕。
        operOrm: async (userId, tableName, methodName, input) => { //procOrm的輸入為: userId, tableName, methodName, input
            let caller = await access.callerOf(userId)
            let r = await procOrm(userId, tableName, methodName, input)
            return await access.scopeSelect(caller, tableName, methodName, r)
        },
        tableNamesExec,
        methodsExec,
        authorizeExec: createAuthorizeExec({ methodsExec, srLog }), //w-serv-webdata 1.1.7 起 useDbOrm=true 時必填; 只放行已識別使用者之 select, 拒絕記 srLog (spec D17)
        tableNamesSync,
        kpFunExt: { //接收參數第1個為userId(getUserIdByToken 驗證後之執行身分), 之後才是前端給予參數 (皆經 wrapErrLog); 一律經 gf(角色與成員檢查, access.mjs), 經 kp 取得呼叫者角色(getUserIdByToken 登錄)

            getWebInfor: kp('getWebInfor', (c) => gf.getWebInfor(c)),

            getChannelsList: kp('getChannelsList', (c) => gf.getChannelsList(c)), //只回呼叫者有權看之頻道; 系統管理者全部
            //清單級 API (後台整表編輯頁一次提交整份清單; opt 只取 baseline 一鍵: 載入時之基準, 不符即整次拒收零寫入); 管理類限系統管理者
            saveChannels: kp('saveChannels', (c, rows, opt) => gf.saveChannels(c, rows, opt)),
            deleteChannels: kp('deleteChannels', (c, ids, opt) => gf.deleteChannels(c, ids, opt)),
            //單筆 API (保留供外部呼叫, 屬不帶基準之寫入; 前端不再呼叫); 管理類限系統管理者
            saveChannel: kp('saveChannel', (c, row) => gf.saveChannel(c, row)),
            deleteChannel: kp('deleteChannel', (c, id) => gf.deleteChannel(c, id)),

            getChannelMembers: kp('getChannelMembers', (c, channelId) => gf.getChannelMembers(c, channelId)), //系統管理者或該頻道成員
            saveChannelMembers: kp('saveChannelMembers', (c, channelId, rows, opt) => gf.saveChannelMembers(c, channelId, rows, opt)), //管理類限系統管理者
            deleteChannelMembers: kp('deleteChannelMembers', (c, channelId, ids, opt) => gf.deleteChannelMembers(c, channelId, ids, opt)),
            saveChannelMember: kp('saveChannelMember', (c, row) => gf.saveChannelMember(c, row)),
            deleteChannelMember: kp('deleteChannelMember', (c, id) => gf.deleteChannelMember(c, id)),

            getMessages: kp('getMessages', (c, channelId, n, afterId) => gf.getMessages(c, channelId, n, afterId)),
            postMessage: kp('postMessage', (c, channelId, content, opt) => gf.postMessage(c, channelId, content, opt)),

            getTasks: kp('getTasks', (c, channelId, n, state, order) => gf.getTasks(c, channelId, n, state, order)),
            getArchivedTasks: kp('getArchivedTasks', (c, channelId, n, order) => gf.getArchivedTasks(c, channelId, n, order)),
            getArchivedMessages: kp('getArchivedMessages', (c, channelId, n) => gf.getArchivedMessages(c, channelId, n)),
            getTask: kp('getTask', (c, taskId) => gf.getTask(c, taskId)),
            claimTask: kp('claimTask', (c, taskId) => gf.claimTask(c, taskId)),
            respondTask: kp('respondTask', (c, taskId, result, state, errorKey, senderType) => gf.respondTask(c, taskId, result, state, errorKey, senderType)),
            resetTask: kp('resetTask', (c, taskId) => gf.resetTask(c, taskId)),
            ackChannel: kp('ackChannel', (c, channelId, lastMessageId) => gf.ackChannel(c, channelId, lastMessageId)),

            uploadFile: kp('uploadFile', (c, inp) => gf.uploadFile(c, inp)), //系統管理者或任一頻道成員
            getFileInfo: kp('getFileInfo', (c, id) => gf.getFileInfo(c, id)),
            getStats: kp('getStats', (c) => gf.getStats(c)), //只計呼叫者有權看之頻道

        },
        fpTableTags: 'tableTags-web-task.json', //w-serv-hapi 之新鍵名(舊名 fnTableTags 僅靠相容層生效)
    })


    return instWServHapiServer
}
